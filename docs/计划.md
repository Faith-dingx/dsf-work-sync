### T6 — 创建 src/index.ts（核心守卫逻辑）

**验收**：导出 `name`、`inject`、`apply`；`Config` 通过 `resolveConfig` 在 apply 内手动调用（N2 fix：不导出 Config，运行时兜底）

```typescript
import { resolveConfig, type RawConfig } from './config.ts';
import type { GuardConfig } from './types.ts';
import type { SubagentRun } from '@deepseek-ai/dsh-subagent/src/types.ts'; // N1 fix: SubagentRun 不在公共导出表，走子路径
import type { SubagentRunEndInfo } from '@deepseek-ai/dsh-subagent'; // 此条保留，公共导出表中已有
import type { Context } from '@deepseek-ai/cordis'; // M3 fix

const PACKAGE_NAME = '@deepseek-ai/dsh-subagent-guard';

export const name = 'subagent-guard';
export const inject = ['subagents', 'sessions', 'logger']; // M3 fix：去掉 'effect' 和 'on'

// H5 fix：模块级标志，防止热重载双重 wrap
let _guardInstalled = false;

export function apply(ctx: Context, rawConfig: RawConfig = {}): void {
  // L3(a) fix：enabled:false 时直接 bypass，不做任何拦截
  const config = resolveConfig(rawConfig);
  if (!config.enabled) {
    ctx.logger.info('[subagent-guard] disabled, bypass');
    return;
  }

  // H5 fix：guard 前 bind originalStart（必须在 guard 检查之前完成）
  const originalStart = ctx.subagents.start.bind(ctx.subagents);

  // H5 fix：模块级防重入
  if (_guardInstalled) return;
  _guardInstalled = true;

  let currentActive = 0;
  // H1 fix：per-run meta，lastSeenCount / lastEventTime 存在每个 run 的 meta 中
  const activeRuns = new Map<
    string,
    {
      startedAt: number;
      run: SubagentRun;
      lastSeenCount: number;
      lastEventTime: number;
    }
  >();

  // M4 fix：end 监听器在 apply 内注册一次（不是每次 start）
  // H0 fix：用 info.id（SessionId = run.id）匹配，不用 info.runId（随机 UUID）
  const onEnd = (info: SubagentRunEndInfo) => {
    if (activeRuns.has(info.id)) {
      activeRuns.delete(info.id);
      currentActive--;
    }
  };
  ctx.on('subagent/end', onEnd);

  ctx.subagents.start = async (name, request) => {
    if (currentActive >= config.maxConcurrent) {
      ctx.logger.warn(`[subagent-guard] 拒绝：子代理并发已达上限（${config.maxConcurrent}）`);
      throw new Error(`当前活跃子代理已达上限（${config.maxConcurrent}），子代理并发已满`);
    }

    // H3 fix：try/catch 防止 start 失败永久泄漏槽位
    let run: SubagentRun;
    try {
      currentActive++;
      run = await originalStart(name, request);
    } catch (e) {
      currentActive--;
      throw e;
    }

    // H0 fix：以 run.id（SessionId）为键存入 activeRuns
    activeRuns.set(run.id, {
      startedAt: Date.now(),
      run,
      lastSeenCount: -1, // H1 fix：初始化 per-run
      lastEventTime: 0, // H1 fix：初始化 per-run
    });

    return run;
  };

  // 空闲检测器 fiber
  ctx.effect(function* idleWatchdog() {
    while (true) {
      yield new Promise<void>((res) => setTimeout(res, config.idlePollMs));
      try {
        for (const [runId, meta] of activeRuns) {
          // M2 fix：远端 run（无 local session）超过 idleTimeoutMs 时强杀
          const session = ctx.sessions.get(runId);
          if (!session) {
            if (Date.now() - meta.startedAt > config.idleTimeoutMs) {
              ctx.logger.warn(`[subagent-guard] 释放远端僵尸 run=${runId}`);
              await meta.run.dispose();
              if (activeRuns.has(runId)) {
                activeRuns.delete(runId);
                currentActive--;
              }
            }
            continue;
          }

          // H1 fix：lastSeenCount / lastEventTime 读 self.meta（per-run）
          const events = session.events;
          const currentCount = events.length;

          if (currentCount === meta.lastSeenCount) {
            const idleMs = Date.now() - meta.lastEventTime;
            if (idleMs > config.idleTimeoutMs) {
              ctx.logger.warn(`[subagent-guard] 释放僵尸 run=${runId}`);
              await meta.run.dispose();
              if (activeRuns.has(runId)) {
                activeRuns.delete(runId);
                currentActive--;
              }
            }
          } else {
            meta.lastSeenCount = currentCount;
            meta.lastEventTime = events[currentCount - 1]?.time ?? meta.startedAt;
          }
        }
      } catch (e) {
        ctx.logger.warn(`[subagent-guard] 空闲检测异常: ${e}`);
        // fail-open: 不杀
      }
    }
  }, 'subagent-guard.idleWatchdog');

  // H5 fix：返回 disposer（不是 ctx.on('dispose')）
  return () => {
    ctx.subagents.start = originalStart;
    _guardInstalled = false;
    activeRuns.clear();
    currentActive = 0;
    ctx.off('subagent/end', onEnd);
  };
}
```

**关键核实结果**（由 T0 填写，2026-08-24 执行核实）：

- `SubagentRun.id` = `SessionId`（等于 child session id）；接口位于 `packages/subagent/subagent/src/types.ts:270`
- `SubagentRunEndInfo.id` = `SessionId`（等于 `run.id`），`SubagentRunEndInfo.runId` = `SubagentRunId`（随机 UUID）；start/end 配对用 `runId`，匹配 activeRuns 用 `id`（lifecycle.ts `observeRun` 确认）
- `ctx.sessions.get(sessionId)` → `Session | undefined`（`packages/core/session/src/index.ts:1055`）
- `Session.events` → `readonly SessionEvent[]`，事件字段为 `seq` + `time: number`（Unix ms，非 `timestamp`；`session/src/types.ts:414`）
- `.ts` 导入可行：`tsconfig.base.json` 有 `allowImportingTsExtensions` + `rewriteRelativeImportExtensions`；`@deepseek-ai/dsh-subagent` package.json `exports["./src/*"]` 存在（N1 通过）
- `Context` 可从 `@deepseek-ai/cordis` 根导入（vendor/cordis/src/context.ts 声明 interface+class；N5 通过）
- 服务名确认：`subagents`（SubagentRuntime）、`sessions`（SessionStore）
- ⚠️ **执行中发现的两处 API 偏差**（实现时修正，测试全绿）：
  1. **cordis 无 `ctx.off`**：`ctx.on()` 返回 disposer（vendor/cordis/src/events.ts:288）；disposer 用 `offEnd()`
  2. **cordis `Disposable` 是函数类型** `() => T`（fiber.ts:74）：effect 返回可调用函数，disposer 用 `void watchdog()`；watchdog 须为 **async generator**（`await` dispose 需要），且 `Effect` 要求 yield `Disposable`，`yield promise` 需 cast（async generator 的 yield 会 await 操作数 → tick 阻塞 = 真实轮询节奏）
  3. cordis `ctx.effect` 同步注销：effect 注册于 fiber，disposer 显式调用 `watchdog()` 保证中断
